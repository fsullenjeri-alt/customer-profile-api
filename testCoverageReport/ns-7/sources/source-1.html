 import lombok.Data;
 import lombok.NoArgsConstructor;
 import org.springframework.validation.annotation.Validated;
 
 /**
  * Partial update: every field is optional (null = leave unchanged),
  * but any field that is sent must be valid.
  */
 @Data
 @AllArgsConstructor
 @NoArgsConstructor

 }
